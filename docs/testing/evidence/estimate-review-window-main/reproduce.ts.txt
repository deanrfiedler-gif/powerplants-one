type Sample = {label:string,pid?:number,statements:{sql:string,ms:number}[],wall_ms?:number,items?:number,next_offset?:number|null,lockAcquired?:()=>void};
const context=new AsyncLocalStorage<Sample>();
let timezone='UTC';
const query=pg.Client.prototype.query;
pg.Client.prototype.query=function(...args:any[]) {
  const sample=context.getStore();
  if(!sample) return query.apply(this,args as any);
  sample.pid=(this as any).processID;
  const sql=typeof args[0]==='string'?args[0]:args[0].text;
  const began=performance.now();
  const execute=()=>query.apply(this,args as any) as any;
  const result=sql==='BEGIN'?query.call(this,"SELECT set_config('TimeZone',$1,false)",[timezone]).then(execute):execute();
  return result.then((result:any)=>{
    sample.statements.push({sql,ms:performance.now()-began});
    if(sql==='SELECT 1 FROM ppo.workspaces WHERE id=$1 FOR UPDATE') sample.lockAcquired?.();
    return result;
  });
} as any;
